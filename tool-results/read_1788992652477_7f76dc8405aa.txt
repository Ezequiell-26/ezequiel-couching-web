   158→function ImcCalc() {
   159→  const [peso, setPeso] = React.useState("");
   160→  const [altura, setAltura] = React.useState("");
   161→  const [res, setRes] = React.useState<{ bmi: number; cat: string; min: number; max: number } | null>(null);
   162→
   163→  React.useEffect(() => {
   164→    const p = loadProfile();
   165→    if (p) {
   166→      setPeso(String(p.weightKg));
   167→      setAltura(String(p.heightCm));
   168→    }
   169→  }, []);
   170→
   171→  function calcular(e: React.FormEvent) {
   172→    e.preventDefault();
   173→    const w = toNum(peso);
   174→    const h = toNum(altura);
   175→    if (!Number.isFinite(w) || w <= 0 || !Number.isFinite(h) || h < 80 || h > 250) {
   176→      errorToast("Peso en kg (p. ej. 75) y altura en cm (p. ej. 178).");
   177→      return;
   178→    }
   179→    const v = imc(w, h);
   180→    const r = idealWeightRange(h);
   181→    setRes({ bmi: v, cat: imcCategory(v), min: r.min, max: r.max });
   182→    track("calc_use", { name: "imc" });
   183→  }
   184→
   185→  return (
   186→    <Card>
   187→      <CardHeader>
   188→        <CardTitle>Índice de masa corporal (IMC)</CardTitle>
   189→        <CardDescription>Clasificación de la OMS: peso en kg respecto a tu altura en metros al cuadrado.</CardDescription>
   190→      </CardHeader>
   191→      <CardContent>
   192→        <form onSubmit={calcular} className="grid gap-4 sm:grid-cols-2">
   193→          <NumberField id="imc-peso" label="Peso" unit="kg" value={peso} onChange={setPeso} min={20} max={400} placeholder="75" />
   194→          <NumberField id="imc-altura" label="Altura" unit="cm" value={altura} onChange={setAltura} min={80} max={250} placeholder="178" />
   195→          <div className="sm:col-span-2">
   196→            <Button type="submit" size="lg">Calcular IMC</Button>
   197→          </div>
   198→        </form>
   199→
   200→        {res ? (
   201→          <CalcResult>
   202→            <p className="text-sm text-muted-foreground">Tu IMC</p>
   203→            <p className="text-4xl font-bold tracking-tight">{fmt1(res.bmi)}</p>
   204→            <div className="mt-2">
   205→              <Badge>{res.cat}</Badge>
   206→            </div>
   207→            <p className="mt-3 text-sm text-muted-foreground">
   208→              Peso saludable para tu altura: <span className="font-medium text-foreground">{res.min}–{res.max} kg</span> (IMC 18,5–24,9).
   209→            </p>
   210→          </CalcResult>
   211→        ) : null}
   212→
   213→        <LimitationNote>
   214→          Estimación de cribado (OMS): no distingue masa muscular de grasa ni distribución corporal. No es un diagnóstico.
   215→        </LimitationNote>
   216→      </CardContent>
   217→    </Card>
   218→  );
   219→}

   223→function TdeeCalc() {
   224→  const [sex, setSex] = React.useState<Sex>("hombre");
   225→  const [age, setAge] = React.useState("");
   226→  const [altura, setAltura] = React.useState("");
   227→  const [peso, setPeso] = React.useState("");
   228→  const [activity, setActivity] = React.useState<ActivityKey>("moderado");
   229→  const [grasa, setGrasa] = React.useState("");
   230→  const [res, setRes] = React.useState<{ bmr: number; tdeeVal: number; katch: number | null } | null>(null);
   231→
   232→  React.useEffect(() => {
   233→    const p = loadProfile();
   234→    if (p) {
   235→      setSex(p.sex);
   236→      setAge(String(p.age));
   237→      setAltura(String(p.heightCm));
   238→      setPeso(String(p.weightKg));
   239→      setActivity(p.activity);
   240→    }
   241→  }, []);
   242→
   243→  function calcular(e: React.FormEvent) {
   244→    e.preventDefault();
   245→    const w = toNum(peso);
   246→    const h = toNum(altura);
   247→    const a = toNum(age);
   248→    const bf = grasa.trim() === "" ? null : toNum(grasa);
   249→    if (!Number.isFinite(w) || w <= 0 || !Number.isFinite(h) || h < 80 || h > 250 || !Number.isFinite(a) || a < 10 || a > 100) {
   250→      errorToast("Edad (10–100), altura (80–250 cm) y peso (kg) válidos.");
   251→      return;
   252→    }
   253→    if (bf !== null && (!Number.isFinite(bf) || bf < 3 || bf > 60)) {
   254→      errorToast("El % de grasa corporal debe estar entre 3 y 60.");
   255→      return;
   256→    }
   257→    const bmr = bmrMifflin(sex, w, h, a);
   258→    setRes({
   259→      bmr,
   260→      tdeeVal: tdee(bmr, activity),
   261→      katch: bf !== null ? bmrKatchMcArdle(w, bf) : null,
   262→    });
   263→    track("calc_use", { name: "calorias" });
   264→  }
   265→
   266→  function guardarPerfil() {
   267→    const w = toNum(peso);
   268→    const h = toNum(altura);
   269→    const a = toNum(age);
   270→    if (!Number.isFinite(w) || w <= 0 || !Number.isFinite(h) || h < 80 || h > 250 || !Number.isFinite(a) || a < 10 || a > 100) {
   271→      errorToast("Completa correctamente edad, altura y peso para guardar el perfil.");
   272→      return;
   273→    }
   274→    const prev = loadProfile();
   275→    saveProfile({
   276→      sex,
   277→      age: a,
   278→      heightCm: h,
   279→      weightKg: w,
   280→      activity,
   281→      goal: prev?.goal ?? "mantener",
   282→      ...(prev?.targetKcal !== undefined ? { targetKcal: prev.targetKcal } : {}),
   283→      updatedAt: Date.now(),
   284→    });
   285→    toast({ title: "Perfil guardado", description: "Las demás herramientas usarán estos datos." });
   286→  }
   287→
   288→  return (
   289→    <Card>
   290→      <CardHeader>
   291→        <CardTitle>Calorías diarias (BMR + TDEE)</CardTitle>
   292→        <CardDescription>
   293→          Metabolismo basal con Mifflin-St Jeor (1990) y gasto total (BMR × factor de actividad).
   294→        </CardDescription>
   295→      </CardHeader>
   296→      <CardContent>
   297→        <form onSubmit={calcular} className="grid gap-4 sm:grid-cols-2">
   298→          <SexField id="tdee-sex" value={sex} onChange={setSex} />
   299→          <NumberField id="tdee-edad" label="Edad" unit="años" value={age} onChange={setAge} min={10} max={100} placeholder="30" />
   300→          <NumberField id="tdee-altura" label="Altura" unit="cm" value={altura} onChange={setAltura} min={80} max={250} placeholder="178" />
   301→          <NumberField id="tdee-peso" label="Peso" unit="kg" value={peso} onChange={setPeso} min={20} max={400} placeholder="75" />
   302→          <NumberField
   303→            id="tdee-grasa"
   304→            label="% grasa corporal (opcional)"
   305→            unit="%"
   306→            value={grasa}
   307→            onChange={setGrasa}
   308→            min={3}
   309→            max={60}
   310→            placeholder="p. ej. 20"
   311→          />
   312→          <ActivityField id="tdee-actividad" value={activity} onChange={setActivity} />
   313→          <div className="flex flex-wrap gap-2 sm:col-span-2">
   314→            <Button type="submit" size="lg">Calcular</Button>
   315→            <Button type="button" variant="secondary" size="lg" onClick={guardarPerfil}>
   316→              Usar en las demás herramientas
   317→            </Button>
   318→          </div>
   319→        </form>
   320→
   321→        {res ? (
   322→          <CalcResult>
   323→            <div className="grid gap-4 sm:grid-cols-2">
   324→              <div>
   325→                <p className="text-sm text-muted-foreground">BMR — Mifflin-St Jeor</p>
   326→                <p className="text-3xl font-bold tracking-tight">
   327→                  {res.bmr} <span className="text-sm font-normal text-muted-foreground">kcal/día</span>
   328→                </p>
   329→              </div>
   330→              <div>
   331→                <p className="text-sm text-muted-foreground">TDEE — BMR × actividad</p>
   332→                <p className="text-3xl font-bold tracking-tight text-primary">
   333→                  {res.tdeeVal} <span className="text-sm font-normal text-muted-foreground">kcal/día</span>
   334→                </p>
   335→              </div>
   336→            </div>
   337→            {res.katch !== null ? (
   338→              <p className="mt-4 rounded-lg border border-border/70 bg-background/40 px-3 py-2 text-sm text-muted-foreground">
   339→                Comparativa Katch-McArdle (con tu % grasa):{" "}
   340→                <span className="font-medium text-foreground">{res.katch} kcal/día</span> de BMR.
   341→              </p>
   342→            ) : (
   343→              <p className="mt-4 text-xs text-muted-foreground">
   344→                Añade tu % de grasa corporal para ver también la estimación Katch-McArdle (basada en masa magra).
   345→              </p>
   346→            )}
   347→          </CalcResult>
   348→        ) : null}
   349→
   350→        <LimitationNote>
   351→          Estimación orientativa con fórmulas poblacionales (Mifflin-St Jeor, 1990; Katch-McArdle si indicas % grasa). No
   352→          sustituye una valoración profesional ni un diagnóstico.
   353→        </LimitationNote>
   354→      </CardContent>
   355→    </Card>
   356→  );
   357→}

   361→function MacrosCalc() {
   362→  const [sex, setSex] = React.useState<Sex>("hombre");
   363→  const [age, setAge] = React.useState("");
   364→  const [altura, setAltura] = React.useState("");
   365→  const [peso, setPeso] = React.useState("");
   366→  const [activity, setActivity] = React.useState<ActivityKey>("moderado");
   367→  const [goal, setGoal] = React.useState<MacroGoal>("mantener");
   368→  const [res, setRes] = React.useState<{ kcal: number; proteinG: number; fatG: number; carbG: number } | null>(null);
   369→
   370→  React.useEffect(() => {
   371→    const p = loadProfile();
   372→    if (p) {
   373→      setSex(p.sex);
   374→      setAge(String(p.age));
   375→      setAltura(String(p.heightCm));
   376→      setPeso(String(p.weightKg));
   377→      setActivity(p.activity);
   378→      setGoal(p.goal);
   379→    }
   380→  }, []);
   381→
   382→  function calcular(e: React.FormEvent) {
   383→    e.preventDefault();
   384→    const w = toNum(peso);
   385→    const h = toNum(altura);
   386→    const a = toNum(age);
   387→    if (!Number.isFinite(w) || w <= 0 || !Number.isFinite(h) || h < 80 || h > 250 || !Number.isFinite(a) || a < 10 || a > 100) {
   388→      errorToast("Edad, altura y peso válidos (o guarda tu perfil en la calculadora de calorías).");
   389→      return;
   390→    }
   391→    const kcalBase = tdee(bmrMifflin(sex, w, h, a), activity);
   392→    setRes(macroSplit(kcalBase, goal, w));
   393→    track("calc_use", { name: "macros" });
   394→  }
   395→
   396→  const cards = res
   397→    ? [
   398→        { label: "Proteína", g: res.proteinG, kcal: res.proteinG * 4, bar: "bg-primary" },
   399→        { label: "Grasa", g: res.fatG, kcal: res.fatG * 9, bar: "bg-accent" },
   400→        { label: "Carbohidratos", g: res.carbG, kcal: res.carbG * 4, bar: "bg-secondary" },
   401→      ]
   402→    : [];
   403→  const totKcal = cards.reduce((acc, c) => acc + c.kcal, 0) || 1;
   404→
   405→  return (
   406→    <Card>
   407→      <CardHeader>
   408→        <CardTitle>Reparto de macronutrientes</CardTitle>
   409→        <CardDescription>
   410→          Parte de tu TDEE (Mifflin-St Jeor × actividad) y ajusta según el objetivo: proteína por kg, grasa en % y el resto en
   411→          carbos.
   412→        </CardDescription>
   413→      </CardHeader>
   414→      <CardContent>
   415→        <form onSubmit={calcular} className="grid gap-4 sm:grid-cols-2">
   416→          <SexField id="mac-sex" value={sex} onChange={setSex} />
   417→          <NumberField id="mac-edad" label="Edad" unit="años" value={age} onChange={setAge} min={10} max={100} placeholder="30" />
   418→          <NumberField id="mac-altura" label="Altura" unit="cm" value={altura} onChange={setAltura} min={80} max={250} placeholder="178" />
   419→          <NumberField id="mac-peso" label="Peso" unit="kg" value={peso} onChange={setPeso} min={20} max={400} placeholder="75" />
   420→          <ActivityField id="mac-actividad" value={activity} onChange={setActivity} />
   421→          <div className="space-y-1.5">
   422→            <Label htmlFor="mac-goal">Objetivo</Label>
   423→            <Select id="mac-goal" value={goal} onChange={(e) => setGoal(e.target.value as MacroGoal)} className="h-11">
   424→              <option value="perder">Perder grasa (−400 kcal)</option>
   425→              <option value="mantener">Mantener peso</option>
   426→              <option value="ganar">Ganar músculo (+300 kcal)</option>
   427→            </Select>
   428→          </div>
   429→          <div className="sm:col-span-2">
   430→            <Button type="submit" size="lg">Calcular macros</Button>
   431→          </div>
   432→        </form>
   433→
   434→        {res ? (
   435→          <CalcResult>
   436→            <p className="text-sm text-muted-foreground">Objetivo calórico</p>
   437→            <p className="text-3xl font-bold tracking-tight">
   438→              {res.kcal} <span className="text-sm font-normal text-muted-foreground">kcal/día</span>
   439→            </p>
   440→            <div className="mt-4 grid gap-3 sm:grid-cols-3">
   441→              {cards.map((c) => (
   442→                <div key={c.label} className="rounded-lg border border-border/70 bg-background/40 p-3">
   443→                  <div className="flex items-center gap-2">
   444→                    <span aria-hidden className={cn("size-2.5 rounded-full", c.bar)} />
   445→                    <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">{c.label}</p>
   446→                  </div>
   447→                  <p className="mt-1 text-2xl font-bold">
   448→                    {c.g} <span className="text-sm font-normal text-muted-foreground">g</span>
   449→                  </p>
   450→                  <p className="text-xs text-muted-foreground">≈ {c.kcal} kcal</p>
   451→                </div>
   452→              ))}
   453→            </div>
   454→            <div
   455→              className="mt-4 flex h-3 w-full overflow-hidden rounded-full bg-muted"
   456→              role="img"
   457→              aria-label={`Reparto calórico: proteína ${Math.round((cards[0].kcal / totKcal) * 100)}%, grasa ${Math.round(
   458→                (cards[1].kcal / totKcal) * 100,
   459→              )}%, carbohidratos ${Math.round((cards[2].kcal / totKcal) * 100)}%`}
   460→            >
   461→              {cards.map((c) => (
   462→                <div key={c.label} className={c.bar} style={{ width: `${(c.kcal / totKcal) * 100}%` }} />
   463→              ))}
   464→            </div>
   465→          </CalcResult>
   466→        ) : null}
   467→
   468→        <LimitationNote>
   469→          Estimación de referencia (proteína 1,6–2 g/kg según objetivo, grasa 25–27% de las kcal): ajusta según tu respuesta y
   470→          tolerancia. No es una prescripción dietética.
   471→        </LimitationNote>
   472→      </CardContent>
   473→    </Card>
   474→  );
   475→}

   479→function OneRmCalc() {
   480→  const [peso, setPeso] = React.useState("");
   481→  const [reps, setReps] = React.useState("");
   482→  const [res, setRes] = React.useState<{ epley: number; brzycki: number; average: number } | null>(null);

   484→  React.useEffect(() => {
   485→    const p = loadProfile();

   487→  }, []);

   489→  function calcular(e: React.FormEvent) {

   566→
   567→function BodyFatCalc() {
   568→  const [metodo, setMetodo] = React.useState<"navy" | "deurenberg">("navy");
   569→
   570→  // Navy
   571→  const [sex, setSex] = React.useState<Sex>("hombre");
   572→  const [altura, setAltura] = React.useState("");
   573→  const [cuello, setCuello] = React.useState("");
   574→  const [cintura, setCintura] = React.useState("");
   575→  const [cadera, setCadera] = React.useState("");
   576→
   577→  // Deurenberg
   578→  const [peso, setPeso] = React.useState("");
   579→  const [alturaD, setAlturaD] = React.useState("");
   580→  const [edad, setEdad] = React.useState("");
   581→
   582→  const [res, setRes] = React.useState<{ navy: number | null; deu: number | null } | null>(null);
   583→
   584→  React.useEffect(() => {
   585→    const p = loadProfile();
   586→    if (p) {
   587→      setSex(p.sex);
   588→      setAltura(String(p.heightCm));
   589→      setPeso(String(p.weightKg));
   590→      setAlturaD(String(p.heightCm));
   591→      setEdad(String(p.age));
   592→    }
   593→  }, []);
   594→
   595→  function calcular(e: React.FormEvent) {
   596→    e.preventDefault();
   597→    let navyVal: number | null = null;
   598→    let deuVal: number | null = null;
   599→
   600→    const h = toNum(altura);
   601→    const neck = toNum(cuello);
   602→    const waist = toNum(cintura);
   603→    const hip = sex === "mujer" ? toNum(cadera) : undefined;
   604→    if (Number.isFinite(h) && h > 0 && Number.isFinite(neck) && neck > 0 && Number.isFinite(waist) && waist > 0) {
   605→      const v = bodyFatNavy(sex, h, neck, waist, hip && Number.isFinite(hip) ? hip : undefined);
   606→      if (Number.isFinite(v) && v > 1 && v < 70) navyVal = Math.round(v * 10) / 10;
   607→    }
   608→
   609→    const w = toNum(peso);
   610→    const hD = toNum(alturaD);
   611→    const a = toNum(edad);
   612→    if (Number.isFinite(w) && w > 0 && Number.isFinite(hD) && hD > 0 && Number.isFinite(a) && a >= 10 && a <= 100) {
   613→      const v = bodyFatDeurenberg(imc(w, hD), a, sex);
   614→      if (Number.isFinite(v) && v > 1 && v < 70) deuVal = Math.round(v * 10) / 10;
   615→    }
   616→
   617→    if (navyVal === null && deuVal === null) {
   618→      errorToast("Completa los datos de al menos un método con medidas válidas.");
   619→      return;
   620→    }
   621→    setRes({ navy: navyVal, deu: deuVal });
   622→    track("calc_use", { name: "grasa" });
   623→  }
   624→
   625→  return (
   626→    <Card>
   627→      <CardHeader>
   628→        <CardTitle>Grasa corporal estimada</CardTitle>
   629→        <CardDescription>Dos métodos contrastados: circunferencias US Navy (1984) y estimación desde IMC de Deurenberg (1991).</CardDescription>
   630→      </CardHeader>
   631→      <CardContent>
   632→        <div role="tablist" aria-label="Método de grasa corporal" className="mb-4 flex gap-2">
   633→          {(
   634→            [
   635→              { id: "navy", label: "US Navy (cinta)" },
   636→              { id: "deurenberg", label: "Deurenberg (IMC)" },
   637→            ] as const
   638→          ).map((t) => (
   639→            <button
   640→              key={t.id}
   641→              type="button"
   642→              role="tab"
   643→              aria-selected={metodo === t.id}
   644→              onClick={() => setMetodo(t.id)}
   645→              className={cn(
   646→                "h-10 rounded-full border px-4 text-sm font-medium transition-colors",
   647→                metodo === t.id ? "border-primary bg-primary text-primary-foreground" : "border-border bg-card hover:bg-accent",
   648→              )}
   649→            >
   650→              {t.label}
   651→            </button>
   652→          ))}
   653→        </div>
   654→
   655→        <form onSubmit={calcular} className="grid gap-4 sm:grid-cols-2">
   656→          {metodo === "navy" ? (
   657→            <>
   658→              <SexField id="bf-sex" value={sex} onChange={setSex} />
   659→              <NumberField id="bf-altura" label="Altura" unit="cm" value={altura} onChange={setAltura} min={80} max={250} placeholder="178" />
   660→              <NumberField id="bf-cuello" label="Cuello" unit="cm" value={cuello} onChange={setCuello} min={20} max={70} placeholder="38" />
   661→              <NumberField id="bf-cintura" label="Cintura" unit="cm" value={cintura} onChange={setCintura} min={40} max={200} placeholder="85" />
   662→              {sex === "mujer" ? (
   663→                <NumberField id="bf-cadera" label="Cadera (mujeres)" unit="cm" value={cadera} onChange={setCadera} min={50} max={220} placeholder="98" />
   664→              ) : null}
   665→            </>
   666→          ) : (
   667→            <>
   668→              <SexField id="bf-sex" value={sex} onChange={setSex} />
   669→              <NumberField id="bf-edad" label="Edad" unit="años" value={edad} onChange={setEdad} min={10} max={100} placeholder="30" />
   670→              <NumberField id="bf-peso" label="Peso" unit="kg" value={peso} onChange={setPeso} min={20} max={400} placeholder="75" />
   671→              <NumberField id="bf-altura-d" label="Altura" unit="cm" value={alturaD} onChange={setAlturaD} min={80} max={250} placeholder="178" />
   672→            </>
   673→          )}
   674→          <div className="sm:col-span-2">
   675→            <Button type="submit" size="lg">Calcular grasa estimada</Button>
   676→          </div>
   677→        </form>
   678→
   679→        {res ? (
   680→          <CalcResult>
   681→            <div className="grid gap-4 sm:grid-cols-2">
   682→              {res.navy !== null ? (
   683→                <div>
   684→                  <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">US Navy (Hodgdon &amp; Beckett, 1984)</p>
   685→                  <p className="mt-1 text-3xl font-bold">{fmt1(res.navy)} %</p>
   686→                </div>
   687→              ) : null}
   688→              {res.deu !== null ? (
   689→                <div>
   690→                  <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">Deurenberg (1991, desde IMC)</p>
   691→                  <p className="mt-1 text-3xl font-bold">{fmt1(res.deu)} %</p>
   692→                </div>
   693→              ) : null}
   694→            </div>
   695→            {res.navy === null || res.deu === null ? (
   696→              <p className="mt-3 text-xs text-muted-foreground">
   697→                El otro método se mostrará aquí cuando tengas sus datos: la comparación entre ambos da una idea del margen de error.
   698→              </p>
   699→            ) : null}
   700→          </CalcResult>
   701→        ) : null}
   702→
   703→        <LimitationNote>
   704→          Estimaciones estadísticas: la cinta de medir tiene margen de error y Deurenberg solo criba a partir del IMC. Ninguna es
   705→          un análisis de composición corporal (DEXA, plicometría profesional).
   706→        </LimitationNote>
   707→      </CardContent>
   708→    </Card>
   709→  );
   710→}

   714→function WaterCalc() {
   715→  const [peso, setPeso] = React.useState("");
   716→  const [activity, setActivity] = React.useState<ActivityKey>("moderado");
   717→  const [res, setRes] = React.useState<{ liters: number; glasses: number } | null>(null);
   718→
   719→  React.useEffect(() => {
   720→    const p = loadProfile();
   721→    if (p) {
   722→      setPeso(String(p.weightKg));
   723→      setActivity(p.activity);
   724→    }
   725→  }, []);
   726→
   727→  function calcular(e: React.FormEvent) {
   728→    e.preventDefault();
   729→    const w = toNum(peso);

   731→      errorToast("Introduce un peso válido en kg.");
   732→      return;
   733→    }
   734→    const liters = waterLiters(w, activity);
   735→    setRes({ liters, glasses: Math.round(liters / 0.25) });

   740→    <Card>
   741→      <CardHeader>

   746→        <form onSubmit={calcular} className="grid gap-4 sm:grid-cols-2">

   748→          <ActivityField id="agua-actividad" value={activity} onChange={setActivity} />

   752→        </form>

   754→        {res ? (

   756→            <p className="text-sm text-muted-foreground">Consumo orientativo</p>

   759→            </p>
   760→            <p className="mt-1 text-sm text-muted-foreground">
