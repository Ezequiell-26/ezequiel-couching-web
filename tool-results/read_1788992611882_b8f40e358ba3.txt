    85→function slugify(text: string): string {
    86→  return text
    87→    .toLowerCase()
    88→    .normalize("NFD")
    89→    .replace(/[\u0300-\u036f]/g, "")
    90→    .replace(/[^a-z0-9]+/g, "-")
    91→    .replace(/^-+|-+$/g, "")
    92→    .slice(0, 80);
    93→}

    95→function eurosToCents(input: string): number | null {
    96→  const normalized = input.trim().replace(",", ".");
    97→  if (!normalized) return null;
    98→  const value = Number(normalized);
    99→  if (!Number.isFinite(value) || value < 0) return null;
   100→  return Math.round(value * 100);
   101→}

   269→function LoginForm({ onSuccess }: { onSuccess: () => Promise<void> }) {
   270→  const [password, setPassword] = React.useState("");
   271→  const [error, setError] = React.useState<string | null>(null);
   272→  const [busy, setBusy] = React.useState(false);
   273→
   274→  const handleSubmit = async (e: React.FormEvent) => {
   275→    e.preventDefault();
   276→    if (busy) return;
   277→    setError(null);
   278→    setBusy(true);
   279→    try {
   280→      const res = await fetch("/api/admin/session", {
   281→        method: "POST",
   282→        headers: { "Content-Type": "application/json" },
   283→        body: JSON.stringify({ password }),
   284→      });
   285→      if (res.ok) {
   286→        setPassword("");
   287→        await onSuccess();
   288→        return;
   289→      }
   290→      const message = await readError(res);
   291→      setError(res.status === 429 ? message : res.status === 401 ? "Contraseña incorrecta." : message);
   292→    } catch {
   293→      setError("Error de conexión. Inténtalo de nuevo.");
   294→    } finally {
   295→      setBusy(false);
   296→    }
   297→  };
   298→
   299→  return (
   300→    <Card className="mx-auto max-w-md">
   301→      <CardContent className="p-6">
   302→        <h2 className="flex items-center gap-2 text-lg font-semibold">
   303→          <LockKeyhole aria-hidden className="size-5 text-primary" />
   304→          Acceso de administración
   305→        </h2>
   306→        <p className="mt-1 text-sm text-muted-foreground">
   307→          Introduce la contraseña del panel para gestionar el catálogo.
   308→        </p>
   309→        <form onSubmit={handleSubmit} className="mt-5 space-y-4" noValidate>
   310→          <div className="space-y-2">
   311→            <Label htmlFor="admin-password">Contraseña</Label>
   312→            <Input
   313→              id="admin-password"
   314→              type="password"
   315→              autoComplete="current-password"
   316→              value={password}
   317→              onChange={(e) => setPassword(e.target.value)}
   318→              required
   319→              autoFocus
   320→            />
   321→          </div>
   322→          {error ? <FormError message={error} /> : null}
   323→          <Button type="submit" className="w-full" disabled={busy || password.length === 0}>
   324→            {busy ? "Entrando…" : "Entrar"}
   325→          </Button>
   326→        </form>
   327→      </CardContent>
   328→    </Card>
   329→  );
   330→}

   334→function useList<T>(url: string, key: string, onExpired: () => Promise<void>) {
   335→  const [items, setItems] = React.useState<T[]>([]);
   336→  const [state, setState] = React.useState<"loading" | "ready" | "error">("loading");
   337→
   338→  const load = React.useCallback(async () => {
   339→    setState("loading");
   340→    try {
   341→      const res = await fetch(url, { cache: "no-store" });
   342→      if (res.status === 401) {
   343→        await onExpired();
   344→        return;
   345→      }
   346→      if (!res.ok) throw new Error(key);
   347→      const data = (await res.json()) as Record<string, unknown>;
   348→      setItems(Array.isArray(data[key]) ? (data[key] as T[]) : []);
   349→      setState("ready");
   350→    } catch {
   351→      setState("error");
   352→    }
   353→  }, [url, key, onExpired]);
   354→
   355→  React.useEffect(() => {
   356→    void load();
   357→  }, [load]);
   358→
   359→  return { items, state, load };
   360→}

   395→function ProductsTab({ onExpired }: { onExpired: () => Promise<void> }) {

   397→  const [dialog, setDialog] = React.useState<{ mode: "create" | "edit"; item?: Product } | null>(null);
   398→  const [busyId, setBusyId] = React.useState<number | null>(null);
   399→
   400→  const toggleActive = async (product: Product) => {
   401→    setBusyId(product.id);
   402→    try {
   403→      const res = await fetch(`/api/admin/products/${product.id}`, {
   404→        method: "PUT",
   405→        headers: { "Content-Type": "application/json" },
   406→        body: JSON.stringify({ active: !product.active }),
   407→      });
   408→      if (res.status === 401) {
   409→        await onExpired();
   410→        return;
   411→      }
   412→      if (!res.ok) throw new Error("toggle");
   413→      track("admin_action", { action: "product_toggle" });
   414→      toast({ title: "Guardado" });
   415→      await load();
   416→    } catch {
   417→      toast({ title: "Error", description: "No se pudo actualizar el producto.", variant: "error" });
   418→    } finally {
   419→      setBusyId(null);
   420→    }
   421→  };

   424→    setBusyId(product.id);
   425→    try {
   426→      const res = await fetch(`/api/admin/products/${product.id}`, { method: "DELETE" });
   427→      if (res.status === 401) {
   428→        await onExpired();
   429→        return;

   435→        title: "Guardado",
   436→        description: data.soft ? "Producto desactivado (borrado suave)." : "Producto eliminado.",

   439→    } catch {
   440→      toast({ title: "Error", description: "No se pudo eliminar el producto.", variant: "error" });

   442→      setBusyId(null);
   443→    }
   444→  };

   567→function ProductForm({

  1134→      const data = (await res.json()) as { soft?: boolean };
  1135→      track("admin_action", { action: "post_delete" });

  1264→function PostForm({
  1265→  post,
  1266→  onClose,
  1267→  onSaved,
  1268→  onExpired,
  1269→}: {
  1270→  post: Post | null;
  1271→  onClose: () => void;
  1272→  onSaved: () => Promise<void>;
  1273→  onExpired: () => Promise<void>;
  1274→}) {
  1275→  const [title, setTitle] = React.useState(post?.title ?? "");
  1276→  const [slug, setSlug] = React.useState(post?.slug ?? "");
  1277→  const [slugTouched, setSlugTouched] = React.useState(post !== null);
  1278→  const [excerpt, setExcerpt] = React.useState(post?.excerpt ?? "");
  1279→  const [body, setBody] = React.useState(post?.body ?? "");
  1280→  const [category, setCategory] = React.useState(post?.category ?? "general");
  1281→  const [published, setPublished] = React.useState(post?.published ?? true);
  1282→  const [error, setError] = React.useState<string | null>(null);
  1283→  const [busy, setBusy] = React.useState(false);
  1284→
  1285→  const handleTitle = (value: string) => {
  1286→    setTitle(value);
  1287→    if (!slugTouched) setSlug(slugify(value));
  1288→  };
  1289→
  1290→  const handleSubmit = async (e: React.FormEvent) => {
  1291→    e.preventDefault();
  1292→    if (busy) return;
  1293→    setError(null);
  1294→
  1295→    if (!slug.trim()) {
  1296→      setError("El slug es obligatorio.");
  1297→      return;
  1298→    }
  1299→    if (!category.trim()) {
  1300→      setError("La categoría es obligatoria.");
  1301→      return;
  1302→    }
  1303→
  1304→    const payload = {
  1305→      slug: slug.trim(),
  1306→      title: title.trim(),
  1307→      excerpt: excerpt.trim(),
  1308→      body: body.trim(),
  1309→      category: category.trim(),
  1310→      published,
  1311→    };
  1312→
  1313→    setBusy(true);
  1314→    try {

  1316→        method: post ? "PUT" : "POST",
  1317→        headers: { "Content-Type": "application/json" },
  1318→        body: JSON.stringify(payload),
  1319→      });
  1320→      if (res.status === 401) {
  1321→        await onExpired();
  1322→        return;
  1323→      }
  1324→      if (!res.ok) {
  1325→        setError(await readError(res));
  1326→        return;
  1327→      }
  1328→      track("admin_action", { action: post ? "post_update" : "post_create" });
  1329→      await onSaved();
  1330→    } catch {
  1331→      setError("Error de conexión. Inténtalo de nuevo.");
  1332→    } finally {
  1333→      setBusy(false);
  1334→    }
  1335→  };

  1345→          required
  1346→          maxLength={120}
  1347→        />
  1348→      </div>
  1349→
  1350→      <div className="space-y-2">
  1351→        <Label htmlFor="post-slug">Slug</Label>

  1358→          }}
  1359→          required
  1360→          maxLength={80}
  1361→          placeholder="se genera solo desde el título"
  1362→        />
  1363→      </div>
  1364→
  1365→      <div className="space-y-2">

  1369→          value={excerpt}

  1371→          required
  1372→          maxLength={300}
  1373→          className="min-h-16"
  1374→        />
  1375→      </div>
  1376→
  1377→      <div className="space-y-2">
  1378→        <Label htmlFor="post-body">Cuerpo del artículo</Label>

  1387→      </div>

  1389→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
