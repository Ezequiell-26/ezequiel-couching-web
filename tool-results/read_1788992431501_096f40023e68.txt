   241→   241→function LoginForm({ onSuccess }: { onSuccess: () => Promise<void> }) {
   242→   242→  const [password, setPassword] = React.useState("");
   243→   243→  const [error, setError] = React.useState<string | null>(null);
   244→   244→  const [busy, setBusy] = React.useState(false);
   245→   245→
   246→   246→  const handleSubmit = async (e: React.FormEvent) => {
   247→   247→    e.preventDefault();
   248→   248→    if (busy) return;
   249→   249→    setError(null);
   250→   250→    setBusy(true);
   251→   251→    try {
   252→   252→      const res = await fetch("/api/admin/session", {
   253→   253→        method: "POST",
   254→   254→        headers: { "Content-Type": "application/json" },
   255→   255→        body: JSON.stringify({ password }),
   256→   256→      });
   257→   257→      if (res.ok) {
   258→   258→        setPassword("");
   259→   259→        await onSuccess();
   260→   260→        return;
   261→   261→      }
   262→   262→      const message = await readError(res);
   263→   263→      setError(res.status === 429 ? message : res.status === 401 ? "Contraseña incorrecta." : message);
   264→   264→    } catch {
   265→   265→      setError("Error de conexión. Inténtalo de nuevo.");
   266→   266→    } finally {
   267→   267→      setBusy(false);
   268→   268→    }
   269→   269→  };
   270→   270→
   271→   271→  return (
   272→   272→    <Card className="mx-auto max-w-md">
   273→   273→      <CardContent className="p-6">
   274→   274→        <h2 className="flex items-center gap-2 text-lg font-semibold">
   275→   275→          <LockKeyhole aria-hidden className="size-5 text-primary" />
   276→   276→          Acceso del entrenador
   277→   277→        </h2>
   278→   278→        <p className="mt-1 text-sm text-muted-foreground">
   279→   279→          Introduce la contraseña del panel para continuar.
   280→   280→        </p>
   281→   281→        <form onSubmit={handleSubmit} className="mt-5 space-y-4" noValidate>
   282→   282→          <div className="space-y-2">
   283→   283→            <Label htmlFor="coach-password">Contraseña</Label>
   284→   284→            <Input
   285→   285→              id="coach-password"
   286→   286→              type="password"
   287→   287→              autoComplete="current-password"
   288→   288→              value={password}
   289→   289→              onChange={(e) => setPassword(e.target.value)}
   290→   290→              required
   291→   291→              autoFocus
   292→   292→            />
   293→   293→          </div>
   294→   294→          {error ? (
   295→   295→            <p role="alert" className="text-sm font-medium text-destructive">
   296→   296→              {error}
   297→   297→            </p>
   298→   298→          ) : null}
   299→   299→          <Button type="submit" className="w-full" disabled={busy || password.length === 0}>
   300→   300→            {busy ? "Entrando…" : "Entrar"}
   301→   301→          </Button>
   302→   302→        </form>
   303→   303→      </CardContent>
   304→   304→    </Card>
   305→   305→  );
   306→   306→}

   310→   310→function SummaryTab({ onExpired }: { onExpired: () => Promise<void> }) {
   311→   311→  const [counts, setCounts] = React.useState<Counts | null>(null);
   312→   312→  const [state, setState] = React.useState<"loading" | "ready" | "error">("loading");
   313→   313→
   314→   314→  const load = React.useCallback(async () => {
   315→   315→    setState("loading");
   316→   316→    try {
   317→   317→      const res = await fetch("/api/admin/session", { method: "PUT", cache: "no-store" });
   318→   318→      if (res.status === 401) {
   319→   319→        await onExpired();
   320→   320→        return;
   321→   321→      }
   322→   322→      if (!res.ok) throw new Error("counts");
   323→   323→      const data = (await res.json()) as { counts?: Counts };
   324→   324→      setCounts(
   325→   325→        data.counts ?? { leads: 0, unreadMessages: 0, newIntakes: 0, pendingOrders: 0 },
   326→   326→      );
   327→   327→      setState("ready");
   328→   328→    } catch {
   329→   329→      setState("error");
   330→   330→    }
   331→   331→  }, [onExpired]);
   332→   332→
   333→   333→  React.useEffect(() => {
   334→   334→    void load();
   335→   335→  }, [load]);
   336→   336→
   337→   337→  if (state === "error") {
   338→   338→    return <ErrorState onRetry={() => void load()} />;
   339→   339→  }
   340→   340→
   341→   341→  const items: { label: string; value: number | null }[] = [
   342→   342→    { label: "Leads registrados", value: counts?.leads ?? null },
   343→   343→    { label: "Mensajes sin leer", value: counts?.unreadMessages ?? null },

   352→   352→          <p className="text-sm text-muted-foreground">{item.label}</p>

   497→   497→function MessageItem({

   503→   503→  onUpdate: (m: Message) => void;
