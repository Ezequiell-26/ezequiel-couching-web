   241→function LoginForm({ onSuccess }: { onSuccess: () => Promise<void> }) {
   242→  const [password, setPassword] = React.useState("");
   243→  const [error, setError] = React.useState<string | null>(null);
   244→  const [busy, setBusy] = React.useState(false);
   245→
   246→  const handleSubmit = async (e: React.FormEvent) => {
   247→    e.preventDefault();
   248→    if (busy) return;
   249→    setError(null);
   250→    setBusy(true);
   251→    try {
   252→      const res = await fetch("/api/admin/session", {
   253→        method: "POST",
   254→        headers: { "Content-Type": "application/json" },
   255→        body: JSON.stringify({ password }),
   256→      });
   257→      if (res.ok) {
   258→        setPassword("");
   259→        await onSuccess();
   260→        return;
   261→      }
   262→      const message = await readError(res);
   263→      setError(res.status === 429 ? message : res.status === 401 ? "Contraseña incorrecta." : message);
   264→    } catch {
   265→      setError("Error de conexión. Inténtalo de nuevo.");
   266→    } finally {
   267→      setBusy(false);
   268→    }
   269→  };
   270→
   271→  return (
   272→    <Card className="mx-auto max-w-md">
   273→      <CardContent className="p-6">
   274→        <h2 className="flex items-center gap-2 text-lg font-semibold">
   275→          <LockKeyhole aria-hidden className="size-5 text-primary" />
   276→          Acceso del entrenador
   277→        </h2>
   278→        <p className="mt-1 text-sm text-muted-foreground">
   279→          Introduce la contraseña del panel para continuar.
   280→        </p>
   281→        <form onSubmit={handleSubmit} className="mt-5 space-y-4" noValidate>
   282→          <div className="space-y-2">
   283→            <Label htmlFor="coach-password">Contraseña</Label>
   284→            <Input
   285→              id="coach-password"
   286→              type="password"
   287→              autoComplete="current-password"
   288→              value={password}
   289→              onChange={(e) => setPassword(e.target.value)}
   290→              required
   291→              autoFocus
   292→            />
   293→          </div>
   294→          {error ? (
   295→            <p role="alert" className="text-sm font-medium text-destructive">
   296→              {error}
   297→            </p>
   298→          ) : null}
   299→          <Button type="submit" className="w-full" disabled={busy || password.length === 0}>
   300→            {busy ? "Entrando…" : "Entrar"}
   301→          </Button>
   302→        </form>
   303→      </CardContent>
   304→    </Card>
   305→  );
   306→}

   310→function SummaryTab({ onExpired }: { onExpired: () => Promise<void> }) {
   311→  const [counts, setCounts] = React.useState<Counts | null>(null);
   312→  const [state, setState] = React.useState<"loading" | "ready" | "error">("loading");
   313→
   314→  const load = React.useCallback(async () => {
   315→    setState("loading");
   316→    try {
   317→      const res = await fetch("/api/admin/session", { method: "PUT", cache: "no-store" });
   318→      if (res.status === 401) {
   319→        await onExpired();
   320→        return;
   321→      }
   322→      if (!res.ok) throw new Error("counts");
   323→      const data = (await res.json()) as { counts?: Counts };
   324→      setCounts(
   325→        data.counts ?? { leads: 0, unreadMessages: 0, newIntakes: 0, pendingOrders: 0 },
   326→      );
   327→      setState("ready");
   328→    } catch {
   329→      setState("error");
   330→    }
   331→  }, [onExpired]);
   332→
   333→  React.useEffect(() => {
   334→    void load();
   335→  }, [load]);
   336→
   337→  if (state === "error") {
   338→    return <ErrorState onRetry={() => void load()} />;
   339→  }
   340→
   341→  const items: { label: string; value: number | null }[] = [
   342→    { label: "Leads registrados", value: counts?.leads ?? null },
   343→    { label: "Mensajes sin leer", value: counts?.unreadMessages ?? null },
   344→    { label: "Cuestionarios nuevos", value: counts?.newIntakes ?? null },
   345→    { label: "Pedidos pendientes", value: counts?.pendingOrders ?? null },
   346→  ];
   347→
   348→  return (
   349→    <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   350→      {items.map((item) => (
   351→        <Card key={item.label} className="p-6">
   352→          <p className="text-sm text-muted-foreground">{item.label}</p>
   353→          {state === "loading" || item.value === null ? (
   354→            <p aria-hidden className="mt-2 animate-pulse text-4xl font-bold text-muted">—</p>
   355→          ) : (
   356→            <p aria-label={`${item.value}`} className="mt-2 text-4xl font-bold text-primary">
   357→              {item.value}
   358→            </p>
   359→          )}
   360→        </Card>
   361→      ))}
   362→      {state === "loading" ? <span className="sr-only">Cargando resumen…</span> : null}
   363→    </div>
   364→  );
   365→}

   451→function MessagesTab({ onExpired }: { onExpired: () => Promise<void> }) {
   452→  const [messages, setMessages] = React.useState<Message[]>([]);
   453→  const [state, setState] = React.useState<"loading" | "ready" | "error">("loading");
   454→

   458→      const res = await fetch("/api/admin/messages", { cache: "no-store" });
   459→      if (res.status === 401) {
   460→        await onExpired();
   461→        return;
   462→      }
   463→      if (!res.ok) throw new Error("messages");
   464→      const data = (await res.json()) as { messages?: Message[] };
   465→      setMessages(data.messages ?? []);
   466→      setState("ready");
   467→    } catch {
   468→      setState("error");
   469→    }
   470→  }, [onExpired]);
   471→
   472→  React.useEffect(() => {
   473→    void load();
   474→  }, [load]);
   475→
   476→  if (state === "loading") return <LoadingState label="Cargando mensajes…" />;
   477→  if (state === "error") return <ErrorState onRetry={() => void load()} />;
   478→  if (messages.length === 0)
   479→    return <EmptyState title="Sin mensajes todavía." hint="Los mensajes del formulario de contacto aparecerán aquí." />;
   480→
   481→  return (

   484→        <MessageItem
   485→          key={message.id}
   486→          message={message}
   487→          onUpdate={(updated) =>
   488→            setMessages((prev) => prev.map((m) => (m.id === updated.id ? updated : m)))

   490→          onExpired={onExpired}
   491→        />
   492→      ))}
   493→    </div>
   494→  );
   495→}
   496→
   497→function MessageItem({
   498→  message,
   499→  onUpdate,
   500→  onExpired,
   501→}: {
   502→  message: Message;
   503→  onUpdate: (m: Message) => void;
   504→  onExpired: () => Promise<void>;
   505→}) {
   506→  const [reply, setReply] = React.useState(message.reply ?? "");
   507→  const [saving, setSaving] = React.useState<"read" | "reply" | null>(null);
   508→  const [error, setError] = React.useState<string | null>(null);
   509→
   510→  const patch = async (body: { read?: boolean; reply?: string }, kind: "read" | "reply") => {
   511→    setError(null);
   512→    setSaving(kind);
   513→    try {
   514→      const res = await fetch(`/api/admin/messages/${message.id}`, {
   515→        method: "PATCH",
   516→        headers: { "Content-Type": "application/json" },
   517→        body: JSON.stringify(body),
   518→      });
   519→      if (res.status === 401) {
   520→        await onExpired();
   521→        return;
   522→      }
   523→      if (!res.ok) throw new Error("patch");
   524→      const data = (await res.json()) as { message?: Message };
   525→      if (data.message) onUpdate(data.message);
   526→      if (kind === "reply") toast({ title: "Guardado", description: "Nota interna guardada." });
   527→    } catch {
   528→      setError("No se pudo guardar. Inténtalo de nuevo.");
   529→    } finally {
   530→      setSaving(null);
   531→    }
   532→  };
   533→
   534→  return (
   535→    <Card className="p-4">
   536→      <div className="flex flex-wrap items-start justify-between gap-2">
   537→        <div>
   538→          <p className="font-medium">{message.name}</p>
   539→          <p className="text-sm break-words text-muted-foreground">{message.email}</p>
   540→        </div>
   541→        <div className="flex items-center gap-2">
   542→          {!message.read ? <Badge>Nuevo</Badge> : null}
   543→          <span className="text-xs text-muted-foreground">{formatDate(message.createdAt)}</span>
   544→        </div>
   545→      </div>
   546→
   547→      <p className="mt-3 whitespace-pre-wrap text-sm leading-relaxed">{message.body}</p>
   548→
   549→      <div className="mt-4 space-y-2">
   550→        <Label htmlFor={`reply-${message.id}`} className="text-xs uppercase tracking-wide text-muted-foreground">
   551→          Respuesta interna (nota)
   552→        </Label>
   553→        <Textarea
   554→          id={`reply-${message.id}`}
   555→          value={reply}
   556→          onChange={(e) => setReply(e.target.value)}
   557→          placeholder="Nota privada sobre este mensaje…"
   558→          maxLength={4000}
   559→        />

   653→function IntakeItem({
   654→  intake,
   655→  onUpdate,
   656→  onExpired,
   657→}: {
   658→  intake: Intake;
   659→  onUpdate: (i: Intake) => void;
   660→  onExpired: () => Promise<void>;
   661→}) {
   662→  const [saving, setSaving] = React.useState(false);
   663→  const [error, setError] = React.useState<string | null>(null);
   664→
   665→  const changeStatus = async (status: string) => {
   666→    setError(null);
   667→    setSaving(true);
   668→    try {
   669→      const res = await fetch(`/api/admin/intakes/${intake.id}`, {
   670→        method: "PATCH",
   671→        headers: { "Content-Type": "application/json" },
   672→        body: JSON.stringify({ status }),
   673→      });
   674→      if (res.status === 401) {
   675→        await onExpired();
   676→        return;
   677→      }
   678→      if (!res.ok) throw new Error("status");
   679→      const data = (await res.json()) as { intake?: Intake };
   680→      if (data.intake) onUpdate(data.intake);
   681→      toast({ title: "Guardado", description: "Estado actualizado." });
   682→    } catch {
   683→      setError("No se pudo actualizar el estado.");
   684→    } finally {
   685→      setSaving(false);
   686→    }
   687→  };
   688→
   689→  const rows: { label: string; value: string | null }[] = [
   690→    { label: "Edad", value: intake.age !== null ? `${intake.age} años` : null },
   691→    { label: "Altura", value: intake.heightCm !== null ? `${intake.heightCm} cm` : null },
   692→    { label: "Peso", value: intake.weightKg !== null ? `${intake.weightKg} kg` : null },
   693→    { label: "Objetivo", value: intake.goal },
   694→    { label: "Experiencia", value: intake.experience },
   695→    { label: "Lesiones", value: intake.injuries },
   696→    { label: "Mensaje", value: intake.message },
   697→  ];
   698→
   699→  return (
   700→    <Card className="p-4">
   701→      <div className="flex flex-wrap items-start justify-between gap-2">
   702→        <div>
   703→          <p className="font-medium">{intake.name}</p>
   704→          <p className="text-sm break-words text-muted-foreground">{intake.email}</p>
   705→        </div>
   706→        <div className="flex items-center gap-2">
   707→          <Badge variant={statusBadgeVariant(intake.status)}>
   708→            {INTAKE_STATUS.find((s) => s.value === intake.status)?.label ?? intake.status}
   709→          </Badge>
   710→          <span className="text-xs text-muted-foreground">{formatDate(intake.createdAt)}</span>
   711→        </div>
   712→      </div>
   713→
   714→      <dl className="mt-3 grid grid-cols-1 gap-x-6 gap-y-1.5 text-sm sm:grid-cols-2">
   715→        {rows.map((row) => (
   716→          <div key={row.label} className="flex gap-2">
   717→            <dt className="shrink-0 text-muted-foreground">{row.label}:</dt>
   718→            <dd className="min-w-0 break-words">{row.value ?? "—"}</dd>
   719→          </div>
   720→        ))}
   721→      </dl>
   722→
   723→      <div className="mt-4 flex flex-wrap items-center gap-2">
   724→        <Label htmlFor={`status-${intake.id}`} className="text-xs text-muted-foreground">
   725→          Estado:
   726→        </Label>
   727→        <Select
   728→          id={`status-${intake.id}`}
   729→          value={intake.status}
   730→          disabled={saving}
   731→          onChange={(e) => void changeStatus(e.target.value)}
   732→          className="w-44"
   733→        >
   734→          {INTAKE_STATUS.map((s) => (
   735→            <option key={s.value} value={s.value}>
   736→              {s.label}
   737→            </option>
   738→          ))}
   739→        </Select>
   740→        {saving ? <span className="text-xs text-muted-foreground">Guardando…</span> : null}
   741→      </div>
   742→      {error ? (
   743→        <p role="alert" className="mt-2 text-sm font-medium text-destructive">
   744→          {error}
   745→        </p>
   746→      ) : null}
   747→    </Card>
   748→  );
   749→}
